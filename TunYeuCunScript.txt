				mainBtn.BackgroundColor3 = Color3.fromRGB(36, 40, 52)
				valBox.Visible = false
				loopBtn.Visible = false
			end

	-- ================= 2B. BAY MƯỢT =================
	local fly = state.fly

	local function clearFlyParts()
		if fly.bv then pcall(function() fly.bv:Destroy() end) fly.bv = nil end
		if fly.bg then pcall(function() fly.bg:Destroy() end) fly.bg = nil end
	end

	local function restoreFlyNoclip()
		for part, oldVal in pairs(fly.collide) do
			if part and part.Parent then
				pcall(function() part.CanCollide = oldVal end)
			end
			fly.collide[part] = nil
		end
	end

	local function muteStepSounds(char, mute)
		if not char then return end
		if mute then
			for _, d in ipairs(char:GetDescendants()) do
				if d:IsA("Sound") then
					local n = string.lower(d.Name)
					if n:find("running") or n:find("walking") or n:find("step") or n:find("climb") or n:find("land") then
						if fly.sounds[d] == nil then
							fly.sounds[d] = d.Volume
						end
						d.Volume = 0
					end
				end
			end
		else
			for snd, oldVol in pairs(fly.sounds) do
				if snd and snd.Parent then
					pcall(function() snd.Volume = oldVol end)
				end
				fly.sounds[snd] = nil
			end
		end
	end

	local function setFlyStates(hum, enabled)
		if not hum then return end
		if enabled then
			local states = {
				Enum.HumanoidStateType.Running,
				Enum.HumanoidStateType.RunningNoPhysics,
				Enum.HumanoidStateType.Climbing,
				Enum.HumanoidStateType.Jumping,
				Enum.HumanoidStateType.Landed,
			}
			for _, st in ipairs(states) do
				if fly.humStates[st] == nil then
					fly.humStates[st] = hum:GetStateEnabled(st)
				end
				pcall(function() hum:SetStateEnabled(st, false) end)
			end
			pcall(function() hum:ChangeState(Enum.HumanoidStateType.Physics) end)
		else
			for st, oldEnabled in pairs(fly.humStates) do
				pcall(function() hum:SetStateEnabled(st, oldEnabled) end)
				fly.humStates[st] = nil
			end
			if hum:GetState() == Enum.HumanoidStateType.Physics then
				pcall(function() hum:ChangeState(Enum.HumanoidStateType.Freefall) end)
			end
		end
	end

	local function stopFly()
		if fly.conn then
			fly.conn:Disconnect()
			fly.conn = nil
		end
		clearFlyParts()
		restoreFlyNoclip()

		local char = player.Character
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		muteStepSounds(char, false)
		if hum then
			hum.PlatformStand = false
			setFlyStates(hum, false)
		end
	end

	local function ensureFlyRig(root)
		if not fly.bv or not fly.bv.Parent then
			fly.bv = Instance.new("BodyVelocity")
			fly.bv.Name = "TunFlyVel"
			fly.bv.MaxForce = Vector3.new(1e6, 1e6, 1e6)
			fly.bv.P = 5000
			fly.bv.Velocity = Vector3.zero
			fly.bv.Parent = root
		end
		if not fly.bg or not fly.bg.Parent then
			fly.bg = Instance.new("BodyGyro")
			fly.bg.Name = "TunFlyGyro"
			fly.bg.MaxTorque = Vector3.new(1e6, 1e6, 1e6)
			fly.bg.P = 6000
			fly.bg.D = 220
			fly.bg.CFrame = root.CFrame
			fly.bg.Parent = root
		end
	end

	local function applyFlyNoclip(char)
		if not char then return end
		if not (fly.on and fly.noclip) then
			restoreFlyNoclip()
			return
		end
		for _, d in ipairs(char:GetDescendants()) do
			if d:IsA("BasePart") then
				if fly.collide[d] == nil then
					fly.collide[d] = d.CanCollide
				end
				d.CanCollide = false
			end
		end
	end

	local function startFly()
		if state.destroyed then return end
		stopFly()
		local char = player.Character
		local root = char and char:FindFirstChild("HumanoidRootPart")
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if not char or not root or not hum then return end

		setFlyStates(hum, true)
		muteStepSounds(char, true)
		ensureFlyRig(root)
		applyFlyNoclip(char)

		fly.conn = RunService.Heartbeat:Connect(function()
			if state.destroyed or not fly.on then
				stopFly()
				return
			end

			local curChar = player.Character
			local curRoot = curChar and curChar:FindFirstChild("HumanoidRootPart")
			local curHum = curChar and curChar:FindFirstChildOfClass("Humanoid")
			if not curChar or not curRoot or not curHum or curHum.Health <= 0 then return end

			setFlyStates(curHum, true)
			muteStepSounds(curChar, true)
			ensureFlyRig(curRoot)
			applyFlyNoclip(curChar)

			local cam = Workspace.CurrentCamera
			local camCf = cam and cam.CFrame or curRoot.CFrame
			local moveDir = Vector3.zero

			if UserInputService.TouchEnabled and not UserInputService.KeyboardEnabled then
				local md = curHum.MoveDirection
				if md.Magnitude > 0.05 then
					local lookFlat = Vector3.new(camCf.LookVector.X, 0, camCf.LookVector.Z)
					local rightFlat = Vector3.new(camCf.RightVector.X, 0, camCf.RightVector.Z)
					if lookFlat.Magnitude > 1e-4 then lookFlat = lookFlat.Unit end
					if rightFlat.Magnitude > 1e-4 then rightFlat = rightFlat.Unit end

					local forwardAmt = md:Dot(lookFlat)
					local rightAmt = md:Dot(rightFlat)
					moveDir = (camCf.LookVector * forwardAmt) + (rightFlat * rightAmt)
				end
			else
				if UserInputService:IsKeyDown(Enum.KeyCode.W) then
					moveDir += camCf.LookVector
				end
				if UserInputService:IsKeyDown(Enum.KeyCode.S) then
					moveDir -= camCf.LookVector
				end
				if UserInputService:IsKeyDown(Enum.KeyCode.D) then
					moveDir += camCf.RightVector
				end
				if UserInputService:IsKeyDown(Enum.KeyCode.A) then
					moveDir -= camCf.RightVector
				end
				if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
					moveDir += Vector3.new(0, 1, 0)
				end
				if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) or UserInputService:IsKeyDown(Enum.KeyCode.C) then
					moveDir -= Vector3.new(0, 1, 0)
				end
			end

			if moveDir.Magnitude > 1 then
				moveDir = moveDir.Unit
			end

			fly.bv.Velocity = moveDir * fly.speed
			fly.bg.CFrame = CFrame.lookAt(curRoot.Position, curRoot.Position + camCf.LookVector)
		end)
	end

	local function clearRagdoll()
		for _, s in ipairs(rag.sockets) do

		end
		table.clear(rag.sockets)

			rag.motors[m] = nil
		end
	end

	local function applyRagdoll(char, active)
		if rag.conn then
			rag.conn:Disconnect()
			rag.conn = nil
		end

		local hum = char and char:FindFirstChildOfClass("Humanoid")
		local root = char and char:FindFirstChild("HumanoidRootPart")

		if state.destroyed or not active then
			clearRagdoll()
			if hum then
				pcall(function()
					hum.PlatformStand = false
					hum:ChangeState(Enum.HumanoidStateType.GettingUp)
				end)
			end
			return
		end

		if not char or not hum then return end

		local function buildJoints()
			clearRagdoll()
			for _, m in ipairs(char:GetDescendants()) do
				if m:IsA("Motor6D") and m.Part0 and m.Part1 and m.Part1.Name ~= "Handle" then
					rag.motors[m] = m.Enabled
					m.Enabled = false

					local a0 = Instance.new("Attachment")
					a0.Name = "TunRagAtt0"
					a0.CFrame = m.C0
					a0.Parent = m.Part0

					local a1 = Instance.new("Attachment")
					a1.Name = "TunRagAtt1"
					a1.CFrame = m.C1
					a1.Parent = m.Part1

					local bsc = Instance.new("BallSocketConstraint")
					bsc.Name = "TunRagSocket"
					bsc.Attachment0 = a0
					bsc.Attachment1 = a1
					bsc.LimitsEnabled = true
					bsc.TwistLimitsEnabled = true
					bsc.UpperAngle = 45
					bsc.TwistLowerAngle = -30
					bsc.TwistUpperAngle = 30
					bsc.Parent = m.Part1

					table.insert(rag.sockets, a0)
					table.insert(rag.sockets, a1)
					table.insert(rag.sockets, bsc)
				end
			end
		end

		buildJoints()
		pcall(function()
			hum.PlatformStand = true
			hum:ChangeState(Enum.HumanoidStateType.Physics)
		end)
		if root then
			pcall(function()
				root.AssemblyLinearVelocity = root.CFrame.LookVector * 14 + Vector3.new(0, 8, 0)
				root.AssemblyAngularVelocity = Vector3.new(4, 6, 4)
			end)
		end

	local function parseColorSpec(str, defaultCol)
		local s = string.lower((str or ""):match("^%s*(.-)%s*$"))
		if s == "" then return defaultCol, false, false end
		if s == "cau vong" or s == "cầu vồng" or s == "rainbow" or s == "rgb" then
			return defaultCol, true, true
		end
		local map = {
			["do"] = Color3.fromRGB(255, 65, 65),
			["đỏ"] = Color3.fromRGB(255, 65, 65),
			["red"] = Color3.fromRGB(255, 65, 65),
			["xanh"] = Color3.fromRGB(0, 255, 120),
			["xanh la"] = Color3.fromRGB(0, 255, 120),
			["xanh lá"] = Color3.fromRGB(0, 255, 120),
			["lá"] = Color3.fromRGB(0, 255, 120),
			["green"] = Color3.fromRGB(0, 255, 120),
			["duong"] = Color3.fromRGB(60, 150, 255),
			["dương"] = Color3.fromRGB(60, 150, 255),
			["xanh dương"] = Color3.fromRGB(60, 150, 255),
			["xanh duong"] = Color3.fromRGB(60, 150, 255),
			["blue"] = Color3.fromRGB(60, 150, 255),
			["cyan"] = Color3.fromRGB(0, 225, 255),
			["vang"] = Color3.fromRGB(255, 205, 0),
			["yellow"] = Color3.fromRGB(255, 205, 0),
			["tim"] = Color3.fromRGB(190, 90, 255),
			["tím"] = Color3.fromRGB(190, 90, 255),
			["purple"] = Color3.fromRGB(190, 90, 255),
			["cam"] = Color3.fromRGB(255, 130, 0),
			["orange"] = Color3.fromRGB(255, 130, 0),
			["trang"] = Color3.fromRGB(245, 245, 245),
			["trắng"] = Color3.fromRGB(245, 245, 245),
			["white"] = Color3.fromRGB(245, 245, 245),
			["hong"] = Color3.fromRGB(255, 0, 238),
			["hồng"] = Color3.fromRGB(255, 0, 238),
			["pink"] = Color3.fromRGB(255, 0, 238),
			["black"] = Color3.fromRGB(0, 0, 0),
			["den"] = Color3.fromRGB(0, 0, 0),
			["đen"] = Color3.fromRGB(0, 0, 0),
		}
		if map[s] then return map[s], false, true end
		local r, g, b = s:match("^(%d+)%s*,%s*(%d+)%s*,%s*(%d+)$")
		if r and g and b then
			return Color3.fromRGB(
				math.clamp(tonumber(r) or 0, 0, 255),
				math.clamp(tonumber(g) or 0, 0, 255),
				math.clamp(tonumber(b) or 0, 0, 255)
			), false, true
		end
		return defaultCol, false, false
	end

	local function formatStuds(dist, decimals)
		local d = math.clamp(math.floor(decimals or 2), 0, 10)
		if d <= 0 then
			return tostring(math.round(dist))
		end
		local formatted = string.format("%." .. tostring(d) .. "f", dist)
		return (formatted:gsub("%.", ","))
	end

	local function getTeamColor(targetPlayer)
		local isAlly = false
		if player.Team ~= nil and targetPlayer.Team ~= nil then
			isAlly = (player.Team == targetPlayer.Team)
		elseif player.TeamColor and targetPlayer.TeamColor and targetPlayer.TeamColor ~= BrickColor.White() then
			isAlly = (player.TeamColor == targetPlayer.TeamColor)
		end
		if isAlly then
			return Color3.fromRGB(60, 150, 255)
		end
		return Color3.fromRGB(255, 65, 65)
	end

	local function isSameTeam(targetPlayer)
		if not targetPlayer or targetPlayer == player then return true end

			return player.Team == targetPlayer.Team
		elseif player.TeamColor and targetPlayer.TeamColor and targetPlayer.TeamColor ~= BrickColor.White() then
			return player.TeamColor == targetPlayer.TeamColor
		end
		return false
	end

	local function matchObj(inst, raw)
		local q = string.lower((raw or ""):match("^%s*(.-)%s*$"))
		if q == "" then return false end
		if Players:GetPlayerFromCharacter(inst) or Players:GetPlayerFromCharacter(inst.Parent) then
			return false
		end
		local mode, val = q:match("^(.-)%s*:%s*(.+)$")
		if mode and val then
			if mode == "ten" or mode == "tên" or mode == "name" then
				return string.find(string.lower(inst.Name), val, 1, true) ~= nil
			elseif mode == "loai" or mode == "loại" or mode == "class" then
				return string.find(string.lower(inst.ClassName), val, 1, true) ~= nil
			elseif mode == "nhom" or mode == "nhóm" or mode == "group" or mode == "model" then
				local m = inst:FindFirstAncestorOfClass("Model")
				return m ~= nil and string.find(string.lower(m.Name), val, 1, true) ~= nil
			end
		end
		local m = inst:FindFirstAncestorOfClass("Model")
		return (string.find(string.lower(inst.Name), q, 1, true) ~= nil)
			or (m ~= nil and string.find(string.lower(m.Name), q, 1, true) ~= nil)
	end

	local function evaluatePlayer(p)
		if state.destroyed or not esp.on or not esp.players or p == player then
			removeEntry(p)
			return
		end
		local char = p.Character
		if not char or not char.Parent then
			removeEntry(p)
			return
		end

		local q = string.lower((esp.pFilter or ""):match("^%s*(.-)%s*$"))
		local okName = (q == "")
			or string.find(string.lower(p.Name), q, 1, true)
			or string.find(string.lower(p.DisplayName), q, 1, true)
		if not okName then
			removeEntry(p)
			return
		end

		local root = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChildWhichIsA("BasePart")
		local hum = char:FindFirstChildOfClass("Humanoid")
		if root then
			local chosenStatic = esp.pCustom and esp.pCol or (esp.showTeam and getTeamColor(p) or esp.pCol)
			upsertTarget(p, char, root, p.DisplayName or p.Name, esp.pRainbow, chosenStatic, p, hum)
		end
	end

	local function evaluateObject(obj)
		if state.destroyed or not esp.on or not esp.objects or not obj or not obj.Parent then
			removeEntry(obj)
			return
		end
		if not (obj:IsA("BasePart") or obj:IsA("Model")) then
			return
		end
		if not matchObj(obj, esp.oFilter) then
			removeEntry(obj)
			return
		end

		local adornee = obj:IsA("Model") and (obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart") or obj) or obj
		local part = adornee:IsA("BasePart") and adornee or obj:FindFirstChildWhichIsA("BasePart", true)
		if part then
			local modelAncestor = obj:FindFirstAncestorOfClass("Model")
			local displayTitle = obj:IsA("Model") and obj.Name or (modelAncestor and (modelAncestor.Name .. "/" .. obj.Name) or obj.Name)
			local hum = obj:IsA("Model") and obj:FindFirstChildOfClass("Humanoid") or nil
			upsertTarget(obj, adornee, part, displayTitle, esp.oRainbow, esp.oCol, nil, hum)
		end
	end

	local function handleRenamedInstance(inst)
		if state.destroyed or not esp.autoUpdate or not esp.on then return end
		if not inst or not inst.Parent then
			removeEntry(inst)
			return
		end

		if inst:IsA("BasePart") or inst:IsA("Model") then
			evaluateObject(inst)
		end

		for _, desc in ipairs(inst:GetDescendants()) do
			if desc:IsA("BasePart") or desc:IsA("Model") then
				evaluateObject(desc)
			end
		end
	end

	local function watchInstanceName(inst)
		if state.destroyed or not esp.autoUpdate or not esp.on then return end
		if not (inst:IsA("BasePart") or inst:IsA("Model")) then return end
		if esp.nameWatchers[inst] then return end

		esp.nameWatchers[inst] = inst:GetPropertyChangedSignal("Name"):Connect(function()
			if state.destroyed then return end
			handleRenamedInstance(inst)
		end)
	end

	local function syncEspTargets()
		if state.destroyed or not esp.on then
			clearAllEntries()
			return
		end

		local validKeys = {}

		if esp.players then
			for _, p in ipairs(Players:GetPlayers()) do
				evaluatePlayer(p)
				if esp.entries[p] then
					validKeys[p] = true
				end
			end
		end

		if esp.objects then
			for _, obj in ipairs(Workspace:GetDescendants()) do
				if (obj:IsA("BasePart") or obj:IsA("Model")) and matchObj(obj, esp.oFilter) then
					evaluateObject(obj)
					if esp.entries[obj] then
						validKeys[obj] = true
					end
				end
			end
		end

		for key in pairs(esp.entries) do
			if not validKeys[key] then
				removeEntry(key)
			end
		end
	end

	local function bindAutoListeners()
		disconnectAutoListeners()
		if state.destroyed or not esp.on or not esp.autoUpdate then return end

		table.insert(esp.autoConns, Players.PlayerAdded:Connect(function(p)
			if state.destroyed then return end
			evaluatePlayer(p)
			table.insert(esp.autoConns, p.CharacterAdded:Connect(function()
				task.wait(0.15)
				if not state.destroyed then evaluatePlayer(p) end
			end))
			table.insert(esp.autoConns, p.CharacterRemoving:Connect(function()
				removeEntry(p)
			end))
		end))

		table.insert(esp.autoConns, Players.PlayerRemoving:Connect(function(p)
			removeEntry(p)
		end))

		for _, p in ipairs(Players:GetPlayers()) do
			if p ~= player then
				table.insert(esp.autoConns, p.CharacterAdded:Connect(function()
					task.wait(0.15)
					if not state.destroyed then evaluatePlayer(p) end
				end))
				table.insert(esp.autoConns, p.CharacterRemoving:Connect(function()
					removeEntry(p)
				end))
			end
		end

		for _, desc in ipairs(Workspace:GetDescendants()) do
			if desc:IsA("BasePart") or desc:IsA("Model") then
				watchInstanceName(desc)
			end
		end

		table.insert(esp.autoConns, Workspace.DescendantAdded:Connect(function(desc)
			if state.destroyed then return end
			if desc:IsA("BasePart") or desc:IsA("Model") then
				watchInstanceName(desc)
				evaluateObject(desc)
			end
		end))

		table.insert(esp.autoConns, Workspace.DescendantRemoving:Connect(function(desc)
			if esp.nameWatchers[desc] then
				esp.nameWatchers[desc]:Disconnect()
				esp.nameWatchers[desc] = nil
			end
			if esp.entries[desc] then
				removeEntry(desc)
			end
		end))
	end

	local function refreshEspState()
		syncEspTargets()
		bindAutoListeners()
	end

	local function resolveTargetPart(char)
		if not char then return nil end
		local preferred = {
			["Head"] = { "Head", "HumanoidRootPart", "UpperTorso", "Torso" },
			["HumanoidRootPart"] = { "HumanoidRootPart", "UpperTorso", "Torso", "Head" },
			["UpperTorso"] = { "UpperTorso", "Torso", "HumanoidRootPart", "Head" },
		}
		local list = preferred[aim.partMode] or preferred["Head"]
		for _, partName in ipairs(list) do
			local p = char:FindFirstChild(partName)
			if p and p:IsA("BasePart") then
				return p
			end
		end
		return char:FindFirstChildWhichIsA("BasePart")
	end

	local function isVisibleFromCam(cam, targetPos, targetChar)
		if not aim.wallCheck or not cam then return true end
		local origin = cam.CFrame.Position
		local dir = targetPos - origin
		local dist = dir.Magnitude
		if dist <= 0.1 then return true end

		local params = RaycastParams.new()
		params.FilterType = Enum.RaycastFilterType.Exclude
		local ignoreList = {}
		if player.Character then table.insert(ignoreList, player.Character) end
		if targetChar then table.insert(ignoreList, targetChar) end
		params.FilterDescendantsInstances = ignoreList
		params.IgnoreWater = true

		local res = Workspace:Raycast(origin, dir, params)
		if not res then return true end
		if targetChar and res.Instance and res.Instance:IsDescendantOf(targetChar) then
			return true
		end
		return false
	end

	local function getBestAimTarget(cam)
		if not cam then return nil, nil end
		local origin2D = getAimScreenOrigin(cam)
		local filterLower = string.lower((aim.filter or ""):match("^%s*(.-)%s*$"))

		local bestScore = math.huge
		local bestPart = nil
		local bestPlayer = nil

		for _, p in ipairs(Players:GetPlayers()) do
			if p ~= player then
				local nameMatch = (filterLower == "")
					or string.find(string.lower(p.Name), filterLower, 1, true)
					or string.find(string.lower(p.DisplayName), filterLower, 1, true)

				local teamOk = (not aim.teamCheck) or (not isSameTeam(p))
				local char = p.Character
				local hum = char and char:FindFirstChildOfClass("Humanoid")
				local aliveOk = (not aim.aliveCheck) or (hum and hum.Health > 0 and not char:FindFirstChildOfClass("ForceField"))

				if nameMatch and teamOk and char and char.Parent and aliveOk then
					local part = resolveTargetPart(char)
					if part and part.Parent then
						local aimWorldPos = part.Position
						if aim.predictOn then
							aimWorldPos = aimWorldPos + (part.AssemblyLinearVelocity * aim.predictScale)
						end

						local screenPos, onScreen = worldToGuiPoint(cam, aimWorldPos)
						if onScreen and screenPos.Z > 0 then
							local pt2D = Vector2.new(screenPos.X, screenPos.Y)
							local fovDist = (pt2D - origin2D).Magnitude
							if fovDist <= aim.fovRadius then
								if isVisibleFromCam(cam, aimWorldPos, char) then
									local worldDist = (aimWorldPos - cam.CFrame.Position).Magnitude
									local hpRatio = (hum and hum.MaxHealth > 0) and (hum.Health / hum.MaxHealth) or 1

									local score = fovDist
									if aim.priority == "Distance" then
										score = worldDist
									elseif aim.priority == "LowHP" then
										score = hpRatio * 1000 + fovDist * 0.25
									end

									if score < bestScore then
										bestScore = score
										bestPart = part
										bestPlayer = p
									end
								end
							end
						end
					end
				end
			end
		end

		return bestPart, bestPlayer
	end

	local function shouldRunAimbot()
		if state.destroyed or not aim.on then return false end
		if aim.mode == "SkillClick" and aim.clickQueued then return true end
		if aim.trigMode == "Always" then return true end
		if aim.trigMode == "Toggle" then return aim.toggled end
		return aim.holding
	end

	local function setHudAvatar(targetPlr)
		local hud = state.hud
		if not hud or not hud.avatarImg then return end
		if not targetPlr then
			hud.lastUserId = nil
			hud.avatarImg.Image = ""
			return
		end

		local uid = targetPlr.UserId
		if hud.lastUserId == uid then return end
		hud.lastUserId = uid

		if hud.avatarCache[uid] then
			hud.avatarImg.Image = hud.avatarCache[uid]
			return
		end

		hud.avatarImg.Image = "rbxthumb://type=AvatarHeadShot&id=" .. tostring(uid) .. "&w=150&h=150"

				return Players:GetUserThumbnailAsync(uid, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size100x100)
			end)

				hud.avatarCache[uid] = content
				if not state.destroyed and hud.lastUserId == uid and hud.avatarImg then
					hud.avatarImg.Image = content
				end
			end
		end)
	end

	local function updateTargetHud(cam, targetPart, targetPlr, isLocked)
		local hud = state.hud
		if not hud or not hud.frame then return end
		if state.destroyed or not aim.on or not hud.enabled then
			hud.frame.Visible = false
			return

		if not targetPart or not targetPlr or not targetPart.Parent then
			setHudAvatar(nil)

			hud.hpLbl.Text = "Máu: ---"

		setHudAvatar(targetPlr)

		local refPos = myRoot and myRoot.Position or (cam and cam.CFrame.Position or targetPart.Position)
		local studs = (targetPart.Position - refPos).Magnitude
		hud.distLbl.Text = string.format("%.2f studs", studs)
	end

	local function fireClickAt(x, y)
		local rx = math.round(x)
		local ry = math.round(y)
		pcall(function()
			local absFn = (typeof(mousemoveabs) == "function" and mousemoveabs)
				or (typeof(getgenv) == "function" and typeof(getgenv().mousemoveabs) == "function" and getgenv().mousemoveabs)
			if absFn then
				absFn(rx, ry)
			end
		end)
		pcall(function()
			local vim = game:GetService("VirtualInputManager")
			vim:SendMouseMoveEvent(rx, ry, game)
			vim:SendMouseButtonEvent(rx, ry, 0, true, game, 1)
			task.delay(0.02, function()
				pcall(function()
					vim:SendMouseButtonEvent(rx, ry, 0, false, game, 1)
				end)
			end)
		end)
		if typeof(mouse1click) == "function" then
			pcall(mouse1click)
		end
	end

	local function updateAimbot(dt)
		if state.destroyed or not aim.on then
			if aim.circle then aim.circle.Visible = false end
			if aim.cursorDot then aim.cursorDot.Visible = false end
			if state.hud and state.hud.frame then state.hud.frame.Visible = false end
			return
		end
		local cam = Workspace.CurrentCamera
		if not cam then
			aim.circle.Visible = false
			if state.hud and state.hud.frame then state.hud.frame.Visible = false end
			setAimStatus("Lỗi: Không có Camera")
			return
		end

		local origin2D = getAimScreenOrigin(cam)
		aim.circle.Visible = aim.showFov
		aim.circle.Size = UDim2.new(0, aim.fovRadius * 2, 0, aim.fovRadius * 2)
		aim.circle.Position = UDim2.new(0, origin2D.X, 0, origin2D.Y)

		local targetPart, targetPlr = getBestAimTarget(cam)
		local activeLock = shouldRunAimbot()
		updateTargetHud(cam, targetPart, targetPlr, activeLock)

		if not activeLock then
			aim.target = nil
			if aim.cursorDot then aim.cursorDot.Visible = false end
			aim.stroke.Color = Color3.fromRGB(0, 220, 255)
			if aim.trigMode == "Hold" then
				setAimStatus("Chờ giữ: " .. aim.holdKeyName)
			else
				setAimStatus("Chờ bật: " .. aim.holdKeyName)
			end
			return
		end

		aim.target = targetPlr
		if not targetPart then
			if aim.cursorDot then aim.cursorDot.Visible = false end
			aim.stroke.Color = Color3.fromRGB(0, 220, 255)
			setAimStatus("Không thấy mục tiêu")
			return
		end

		aim.stroke.Color = Color3.fromRGB(255, 75, 75)
		setAimStatus("Đang ngắm: " .. (targetPlr.DisplayName or targetPlr.Name))

		local desiredWorldPos = targetPart.Position
		if aim.predictOn then
			desiredWorldPos = desiredWorldPos + (targetPart.AssemblyLinearVelocity * aim.predictScale)
		end

		local alpha = math.clamp(aim.smooth, 0.02, 1)
		if dt and dt > 0 and alpha < 1 then
			alpha = 1 - ((1 - alpha) ^ (dt * 60))
		end

		if aim.mode == "Cursor" then
			local guiPt, onScreen = worldToGuiPoint(cam, desiredWorldPos)
			if not onScreen or guiPt.Z <= 0.1 then
				if aim.cursorDot then aim.cursorDot.Visible = false end
				return
			end
			local vp = cam.ViewportSize
			local inset = getGuiInsetOffset()
			local minX, maxX = inset.X + 2, inset.X + vp.X - 2
			local minY, maxY = inset.Y + 2, inset.Y + vp.Y - 2
			local targetPt = Vector2.new(
				math.clamp(guiPt.X, minX, maxX),
				math.clamp(guiPt.Y, minY, maxY)
			)
			if aim.cursorDot then
				aim.cursorDot.Visible = true
				aim.cursorDot.Position = UDim2.new(0, targetPt.X, 0, targetPt.Y)
			end
		elseif aim.mode == "SkillClick" then
			local guiPt, onScreen = worldToGuiPoint(cam, desiredWorldPos)
			if not onScreen or guiPt.Z <= 0.1 then
				if aim.cursorDot then aim.cursorDot.Visible = false end
				return
			end
			local vp = cam.ViewportSize
			local inset = getGuiInsetOffset()
			local minX, maxX = inset.X + 2, inset.X + vp.X - 2
			local minY, maxY = inset.Y + 2, inset.Y + vp.Y - 2
			local targetPt = Vector2.new(
				math.clamp(guiPt.X, minX, maxX),
				math.clamp(guiPt.Y, minY, maxY)
			)
			if aim.cursorDot then
				aim.cursorDot.Visible = true
				aim.cursorDot.Position = UDim2.new(0, targetPt.X, 0, targetPt.Y)
			end
			local now = os.clock()
			if aim.clickQueued and (now - aim.lastClickTick) >= 0.08 then
				aim.clickQueued = false
				aim.lastClickTick = now
				fireClickAt(targetPt.X, targetPt.Y)
			end
		elseif aim.mode == "Trigger" then
			local guiPt, onScreen = worldToGuiPoint(cam, desiredWorldPos)
			if onScreen and guiPt.Z > 0 then
				local mousePt = getAimScreenOrigin(cam)
				if (Vector2.new(guiPt.X, guiPt.Y) - mousePt).Magnitude <= math.max(18, aim.fovRadius * 0.25) then
					local now = os.clock()
					if (now - aim.lastClickTick) >= 0.09 then
						aim.lastClickTick = now
						fireClickAt(guiPt.X, guiPt.Y)
					end
				end
			end
		else
			local curCf = cam.CFrame
			local goalCf = CFrame.lookAt(curCf.Position, desiredWorldPos)
			cam.CFrame = alpha >= 1 and goalCf or curCf:Lerp(goalCf, alpha)
		end
	end

	local function applyMouseUnlock(enable)
		if enable and not state.destroyed then
			if sys.savedBehavior == nil then
				pcall(function() sys.savedBehavior = UserInputService.MouseBehavior end)
			end
			if sys.savedIcon == nil then
				pcall(function() sys.savedIcon = UserInputService.MouseIconEnabled end)
			end
			pcall(function()
				UserInputService.MouseIconEnabled = true
				if not UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
					UserInputService.MouseBehavior = Enum.MouseBehavior.Default
				end
			end)
			if not sys.mouseConn then
				sys.mouseConn = RunService.RenderStepped:Connect(function()
					if state.destroyed or not sys.unlockMouse then return end
					if mainFrame.Visible or (sys.overlayBtn and sys.overlayBtn.Visible) then
						pcall(function()
							if not UserInputService.MouseIconEnabled then
								UserInputService.MouseIconEnabled = true
							end
							if not UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
								if UserInputService.MouseBehavior ~= Enum.MouseBehavior.Default then
									UserInputService.MouseBehavior = Enum.MouseBehavior.Default
								end
							end
						end)
					end
				end)
			end
		else
			if sys.mouseConn then
				sys.mouseConn:Disconnect()
				sys.mouseConn = nil
			end
			pcall(function()
				if sys.savedBehavior ~= nil then
					UserInputService.MouseBehavior = sys.savedBehavior
				end
				if sys.savedIcon ~= nil then
					UserInputService.MouseIconEnabled = sys.savedIcon
				end
			end)
			sys.savedBehavior = nil
			sys.savedIcon = nil
		end
	end

	local function findPlayerByQuery(rawQuery)
		local q = string.lower((rawQuery or ""):match("^%s*(.-)%s*$"))
		if q == "" then return nil end
		for _, tier in ipairs({ 1, 2, 3 }) do
			for _, p in ipairs(Players:GetPlayers()) do
				if p ~= player then
					local un = string.lower(p.Name)
					local dn = string.lower(p.DisplayName)
					local match = false
					if tier == 1 then
						match = (un == q or dn == q)
					elseif tier == 2 then
						match = (string.sub(un, 1, #q) == q or string.sub(dn, 1, #q) == q)
					else
						match = (string.find(un, q, 1, true) ~= nil or string.find(dn, q, 1, true) ~= nil)
					end
					if match then
						return p
					end
				end
			end
		end
		return nil
	end

	local function doHopServer()
		if state.destroyed or extra.hopping then return end
		extra.hopping = true
		hopBtn.Text = "Đang tìm server..."
		task.spawn(function()
			local cursor = ""
			local foundJobId = nil
			for _ = 1, 6 do
				if state.destroyed then break end
				local url = string.format(
					"https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100%s",
					game.PlaceId,
					cursor ~= "" and ("&cursor=" .. HttpService:UrlEncode(cursor)) or ""
				)
				local okGet, raw = pcall(function()
					return game:HttpGet(url)
				end)
				if okGet and type(raw) == "string" then
					local okDec, decoded = pcall(function()
						return HttpService:JSONDecode(raw)
					end)
					if okDec and type(decoded) == "table" and type(decoded.data) == "table" then
						for _, srv in ipairs(decoded.data) do
							if type(srv) == "table"
								and type(srv.id) == "string"
								and srv.id ~= game.JobId
								and tonumber(srv.playing)
								and tonumber(srv.maxPlayers)
								and srv.playing < srv.maxPlayers then
								foundJobId = srv.id
								break
							end
						end
						if foundJobId then break end
						cursor = (type(decoded.nextPageCursor) == "string") and decoded.nextPageCursor or ""
						if cursor == "" then break end
					else
						break
					end
				else
					break
				end
			end

			if not state.destroyed then
				if foundJobId then
					hopBtn.Text = "Đang chuyển server..."
					pcall(function()
						TeleportService:TeleportToPlaceInstance(game.PlaceId, foundJobId, player)
					end)
				else
					hopBtn.Text = "Không thấy server khác"
				end
				task.delay(2.5, function()
					extra.hopping = false
					if not state.destroyed then
						hopBtn.Text = "Hop Server"
					end
				end)
			end
		end)
	end

	local function isCharDescendant(part)
		local m = part:FindFirstAncestorOfClass("Model")
		while m do
			if Players:GetPlayerFromCharacter(m) or m:FindFirstChildOfClass("Humanoid") then
				return true
			end
			m = m.Parent and m.Parent:FindFirstAncestorOfClass("Model") or nil
		end
		return false
	end

	local function applyXrayToPart(part)
		if part:IsA("BasePart") and not isCharDescendant(part) and part.Transparency < 0.98 then
			if extra.xraySaved[part] == nil then
				extra.xraySaved[part] = part.LocalTransparencyModifier
			end
			part.LocalTransparencyModifier = math.clamp(extra.xrayAlpha, 0.1, 0.95)
		end
	end

	local function setXrayEnabled(on)
		extra.xray = on
		if extra.xrayConn then
			extra.xrayConn:Disconnect()
			extra.xrayConn = nil
		end
		if state.destroyed or not on then
			for part, oldMod in pairs(extra.xraySaved) do
				if part and part.Parent then
					pcall(function() part.LocalTransparencyModifier = oldMod end)
				end
				extra.xraySaved[part] = nil
			end
			return
		end
		for _, d in ipairs(Workspace:GetDescendants()) do
			applyXrayToPart(d)
		end
		extra.xrayConn = Workspace.DescendantAdded:Connect(function(d)
			if not state.destroyed and extra.xray then
				applyXrayToPart(d)
			end
		end)
	end

	local xrayAlphaBox = makeBox(xrayRow, tostring(extra.xrayAlpha), "Mờ (0.1-0.95)", UDim2.new(0.34, -2, 1, 0), UDim2.new(0.66, 2, 0, 0))
	xrayAlphaBox.TextSize = 10

	end))

	local minBright, maxBright = 0, 10
	local function applyBrightnessNow()
		if not extra.fbOn or state.destroyed then return end
		pcall(function()
			Lighting.Brightness = extra.fbBrightness
			Lighting.ClockTime = 14
			Lighting.FogEnd = 100000
			Lighting.GlobalShadows = false
			Lighting.Ambient = Color3.fromRGB(255, 255, 255)
			Lighting.OutdoorAmbient = Color3.fromRGB(255, 255, 255)
		end)
	end

	local function syncSliderVisual()
		local ratio = math.clamp((extra.fbBrightness - minBright) / (maxBright - minBright), 0, 1)
		sliderFill.Size = UDim2.new(ratio, 0, 1, 0)
		sliderKnob.Position = UDim2.new(ratio, 0, 0.5, 0)
		fbValLbl.Text = string.format("Độ sáng: %.1f", extra.fbBrightness)
	end

	local function setBrightnessValue(val)
		extra.fbBrightness = math.clamp(math.floor(val * 10 + 0.5) / 10, minBright, maxBright)
		syncSliderVisual()
		applyBrightnessNow()
	end

	local function setFullbrightEnabled(on)
		extra.fbOn = on
		if extra.fbConn then
			extra.fbConn:Disconnect()
			extra.fbConn = nil
		end
		if state.destroyed or not on then
			fbToggleBtn.Text = "Fullbright : TẮT"
			fbToggleBtn.BackgroundColor3 = Color3.fromRGB(36, 40, 52)
			if extra.fbSaved then
				local s = extra.fbSaved
				pcall(function()
					Lighting.Brightness = s.Brightness
					Lighting.ClockTime = s.ClockTime
					Lighting.FogEnd = s.FogEnd
					Lighting.GlobalShadows = s.GlobalShadows
					Lighting.Ambient = s.Ambient
					Lighting.OutdoorAmbient = s.OutdoorAmbient
				end)
				extra.fbSaved = nil
			end
			return
		end

		if not extra.fbSaved then
			extra.fbSaved = {
				Brightness = Lighting.Brightness,
				ClockTime = Lighting.ClockTime,
				FogEnd = Lighting.FogEnd,
				GlobalShadows = Lighting.GlobalShadows,
				Ambient = Lighting.Ambient,
				OutdoorAmbient = Lighting.OutdoorAmbient,
			}
		end

		fbToggleBtn.Text = "Fullbright: BẬT"
		fbToggleBtn.BackgroundColor3 = Color3.fromRGB(0, 140, 170)
		applyBrightnessNow()
		extra.fbConn = RunService.Heartbeat:Connect(function()
			if not state.destroyed and extra.fbOn then
				applyBrightnessNow()
			end
		end)
	end

	local function updateSliderFromX(screenX)
		local absPos = sliderTrack.AbsolutePosition.X
		local absSize = sliderTrack.AbsoluteSize.X
		if absSize <= 0 then return end
		local ratio = math.clamp((screenX - absPos) / absSize, 0, 1)
		setBrightnessValue(minBright + ratio * (maxBright - minBright))
	end

			slidingBright = true
			updateSliderFromX(input.Position.X)
		end
	end))

	track(UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			slidingBright = false
		end
	end))

	track(UserInputService.InputChanged:Connect(function(input)
		if state.destroyed or not slidingBright then return end
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			updateSliderFromX(input.Position.X)
		end
	end))

	track(RunService.RenderStepped:Connect(function(dt)
		if state.destroyed then return end

		if not esp.on then
			hideLines()
			return
		end

		local cam = Workspace.CurrentCamera
		local originPos = cam and cam.CFrame.Position or Vector3.zero
		local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
		if root then
			originPos = root.Position
		end

		local lineCount = 0
		local vp = cam and cam.ViewportSize
		local startPt = vp and Vector2.new(vp.X * 0.5, vp.Y * 0.92) or Vector2.zero

		for key, item in pairs(esp.entries) do
			local part = item.part
			if not part or not part.Parent then
				removeEntry(key)
			else
				local currentStatic = item.staticColor
				if item.targetPlayer and item.targetPlayer.Parent and esp.showTeam and not esp.pCustom then
					currentStatic = getTeamColor(item.targetPlayer)
				end

				local col = item.isRainbow and getRainbowColor(item.offset) or currentStatic

				if item.hl and item.hl.Parent then
					item.hl.FillColor = col
					item.hl.OutlineColor = col
					item.hl.FillTransparency = esp.outlineOnly and 1 or 0.65
					item.hl.DepthMode = esp.throughWall and Enum.HighlightDepthMode.AlwaysOnTop or Enum.HighlightDepthMode.Occluded
				end

				if item.bb and item.lbl and item.bb.Parent then
					local showInfo = esp.showName or esp.showStuds or esp.showHp
					item.bb.Enabled = showInfo
					item.bb.AlwaysOnTop = esp.throughWall
					if showInfo then
						item.lbl.TextColor3 = col

						local line1Parts = {}
						if esp.showName then
							table.insert(line1Parts, item.labelName)
						end
						if esp.showHp then
							local hum = item.humanoid
							if not hum or not hum.Parent then
								local parentModel = part:FindFirstAncestorOfClass("Model")
								hum = parentModel and parentModel:FindFirstChildOfClass("Humanoid") or nil
								item.humanoid = hum
							end
							if hum and hum.Parent then
								local hpNow = math.max(0, math.round(hum.Health))
								local hpMax = math.max(1, math.round(hum.MaxHealth))
								table.insert(line1Parts, tostring(hpNow) .. "/" .. tostring(hpMax) .. " HP")
							end
						end

						local line2Text = ""
						if esp.showStuds then
							local dist = (part.Position - originPos).Magnitude
							line2Text = formatStuds(dist, esp.decimals) .. " st"
						end

						local line1Text = table.concat(line1Parts, " · ")
						if line1Text ~= "" and line2Text ~= "" then
							item.lbl.Text = line1Text .. "\n" .. line2Text
						else
							item.lbl.Text = line1Text ~= "" and line1Text or line2Text
						end
					end
				end

				if esp.chain and cam and vp then
					local pos, onScreen = cam:WorldToViewportPoint(part.Position)
					if onScreen and pos.Z > 0 then
						lineCount += 1
						local line = getLine(lineCount)
						local endPt = Vector2.new(pos.X, pos.Y)
						local diff = endPt - startPt
						local mid = (startPt + endPt) * 0.5

						line.Size = UDim2.new(0, diff.Magnitude, 0, 2)
						line.Position = UDim2.new(0, mid.X, 0, mid.Y)
						line.Rotation = math.deg(math.atan2(diff.Y, diff.X))
						line.BackgroundColor3 = col
					end
				end
			end
		end

		if not esp.chain then
			hideLines()
		else
			for i = lineCount + 1, #esp.lines do
				esp.lines[i].Visible = false
			end
		end
	end))

	track(UserInputService.InputBegan:Connect(function(input, gp)

		if input.KeyCode == Enum.KeyCode.RightShift then

		end
	end))
