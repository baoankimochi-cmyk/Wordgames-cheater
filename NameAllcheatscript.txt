local function ClearChildren(frame)
	for _, v in ipairs(frame:GetChildren()) do
		if v:IsA("GuiObject") then
			v:Destroy()
		end
	end
end

local function ParseRaw(raw)
	local data = {}
	local current = nil
	for line in raw:gmatch("[^\r\n]+") do
		line = line:match("^%s*(.-)%s*$")
		if line ~= "" then
			local cat = line:match("^%[(.-)%]$")
			if cat then
				current = cat
				data[current] = data[current] or {}
			elseif current then
				table.insert(data[current], line)
			end
		end
	end
	return data
end

local function FetchRaw()
	local fetchRemote = ReplicatedStorage:FindFirstChild("FetchWordData")
	if fetchRemote and fetchRemote:IsA("RemoteFunction") then
		local ok, res = pcall(function()
			return fetchRemote:InvokeServer()
		end)
		if ok and type(res) == "string" and res ~= "" then
			return res
		end
	end

	local okHttp, resHttp = pcall(function()
		return HttpService:GetAsync(URL)
	end)
	if okHttp and type(resHttp) == "string" and resHttp ~= "" then
		return resHttp
	end

	if typeof(game.HttpGet) == "function" then
		local okExec, resExec = pcall(function()
			return game:HttpGet(URL)
		end)
		if okExec and type(resExec) == "string" and resExec ~= "" then
			return resExec
		end
	end

	return nil
end

local function ShowWords(cat)
	ClearChildren(WordList)
	for i, word in ipairs(Categories[cat] or {}) do
		local lbl = Instance.new("TextLabel")
		lbl.Size = UDim2.new(1, -8, 0, 22)
		lbl.BackgroundColor3 = Color3.fromRGB(32, 50, 62)
		lbl.TextColor3 = Color3.fromRGB(235, 245, 250)
		lbl.TextXAlignment = Enum.TextXAlignment.Left
		lbl.ZIndex = 102
		lbl.Text = "  " .. word
		lbl.LayoutOrder = i
		lbl.Parent = WordList
	end
	task.wait()
	WordList.CanvasSize = UDim2.fromOffset(0, WordLayout.AbsoluteContentSize.Y + 6)
end

local function BuildCategories(filter)
	filter = (filter or ""):lower()
	ClearChildren(CatList)
	local names = {}
	for k in pairs(Categories) do
		if filter == "" or k:lower():find(filter, 1, true) then
			table.insert(names, k)
		end
	end
	table.sort(names, function(a, b)
		return a:lower() < b:lower()
	end)

	for i, name in ipairs(names) do
		local btn = Instance.new("TextButton")
		btn.Size = UDim2.new(1, -8, 0, 28)
		btn.BackgroundColor3 = (Selected == name) and Color3.fromRGB(0, 195, 215) or Color3.fromRGB(32, 50, 62)
		btn.TextColor3 = (Selected == name) and Color3.fromRGB(10, 20, 25) or Color3.fromRGB(235, 245, 250)
		btn.TextXAlignment = Enum.TextXAlignment.Left
		btn.ZIndex = 102
		btn.Text = "  " .. name
		btn.LayoutOrder = i
		btn.Parent = CatList

		btn.MouseButton1Click:Connect(function()
			Selected = name
			BuildCategories(SearchBox.Text)
			ShowWords(name)
			Status.Text = "Selected: " .. name .. " (" .. #Categories[name] .. ")"
		end)
	end
	task.wait()
	CatList.CanvasSize = UDim2.fromOffset(0, CatLayout.AbsoluteContentSize.Y + 6)
end

SearchBox:GetPropertyChangedSignal("Text"):Connect(function()
	BuildCategories(SearchBox.Text)
end)

local function GetAnswerBox()
	local roundUI = PlayerGui:FindFirstChild("RoundUI")
	local hud = roundUI and roundUI:FindFirstChild("Hud")
	local panel = hud and hud:FindFirstChild("Panel")
	local inputBar = panel and panel:FindFirstChild("InputBar")
	return inputBar and inputBar:FindFirstChild("AnswerBox")
end

local function SendWordFast(word, box)
	if box and box:IsA("TextBox") then
		box.Text = word
		box:ReleaseFocus(true)
	else
		local remotes = ReplicatedStorage:FindFirstChild("Remotes")
		local submitRemote = remotes and remotes:FindFirstChild("SubmitWord")
		if submitRemote and submitRemote:IsA("RemoteEvent") then
			submitRemote:FireServer(word)
		end
	end
end

local function DoReload()
	Running = false
	RunId += 1
	RunBtn.Text = "RUN"
	Status.Text = "Fetching..."
	local raw = FetchRaw()
	if not raw then
		Status.Text = "Fetch failed"
		return
	end
	Categories = ParseRaw(raw)
	Selected = nil
	ClearChildren(WordList)
	BuildCategories(SearchBox.Text)
	Status.Text = "Loaded categories"
end

ReloadBtn.MouseButton1Click:Connect(DoReload)

RunBtn.MouseButton1Click:Connect(function()
	if Running then
		Running = false
		RunId += 1
		RunBtn.Text = "RUN"
		Status.Text = "Stopped"
		return
	end

	if not Selected or not Categories[Selected] or #Categories[Selected] == 0 then
		Status.Text = "Pick a category first"
		return
	end

	Running = true
	RunId += 1
	local currentId = RunId
	RunBtn.Text = "STOP"

	task.spawn(function()
		local words = Categories[Selected]
		local box = GetAnswerBox()
		for i, word in ipairs(words) do
			if not Running or currentId ~= RunId then
				return
			end
			box = box or GetAnswerBox()
			Status.Text = string.format("[%d/%d] %s", i, #words, word)
			SendWordFast(word, box)
			task.wait(STEP_DELAY)
		end
		if Running and currentId == RunId then
			Running = false
			RunBtn.Text = "RUN"
			Status.Text = "Done"
		end
	end)
end)

task.spawn(DoReload)
